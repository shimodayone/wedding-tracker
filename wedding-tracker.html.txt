// ===================== UTILS =====================
function daysUntil(target) {
  const now = new Date();
  now.setHours(0,0,0,0);
  const t = new Date(target);
  t.setHours(0,0,0,0);
  return Math.ceil((t - now) / 86400000);
}
function formatDate(d) {
  return `${d.getFullYear()}.${String(d.getMonth()+1).padStart(2,'0')}.${String(d.getDate()).padStart(2,'0')}`;
}
function dateKey(d) {
  return formatDate(d);
}
function today() { return new Date(); }

// ===================== TRACKER =====================
function renderTracker() {
  const now = today();
  const year = trackerMonth.getFullYear();
  const month = trackerMonth.getMonth();

  const firstDay = new Date(year, month, 1).getDay();
  const daysInMonth = new Date(year, month+1, 0).getDate();

  // Stats for this month
  let doneCount = 0, totalDays = 0;
  for (let d = 1; d <= daysInMonth; d++) {
    const dd = new Date(year, month, d);
    if (dd > now) break;
    const dow = dd.getDay();
    if (dow !== 0) totalDays++;
    const key = dateKey(dd);
    if (checkedDays[key]) doneCount++;
  }

  const pct = totalDays > 0 ? Math.round(doneCount/totalDays*100) : 0;

  let html = `
    <div class="month-nav">
      <button class="month-arrow" onclick="changeMonth(-1)">‹</button>
      <div class="month-title">${year}年 ${month+1}月</div>
      <button class="month-arrow" onclick="changeMonth(1)">›</button>
    </div>

    <div class="stats-row">
      <div class="stat-card">
        <div class="stat-num">${doneCount}</div>
        <div class="stat-label">完了日数</div>
      </div>
      <div class="stat-card">
        <div class="stat-num">${totalDays}</div>
        <div class="stat-label">トレ日数</div>
      </div>
      <div class="stat-card">
        <div class="stat-num">${pct}<small style="font-size:16px">%</small></div>
        <div class="stat-label">達成率</div>
      </div>
    </div>

    <div class="section-card" style="margin-bottom:20px;">
      <div style="display:flex; justify-content:space-between; margin-bottom:8px;">
        <span style="font-size:13px; font-weight:500;">今月の達成率</span>
        <span style="font-size:13px; color:var(--rose); font-weight:700;">${pct}%</span>
      </div>
      <div class="progress-wrap"><div class="progress-fill" style="width:${pct}%"></div></div>
    </div>

    <div class="cal-grid">
      ${DAYS_JP.map(d => `<div class="cal-head">${d}</div>`).join('')}
  `;

  // Empty cells before
  for (let i = 0; i < firstDay; i++) {
    const prevDate = new Date(year, month, -firstDay + i + 1);
    html += `<div class="cal-cell other-month">${prevDate.getDate()}</div>`;
  }

  for (let d = 1; d <= daysInMonth; d++) {
    const dd = new Date(year, month, d);
    const dow = dd.getDay();
    const key = dateKey(dd);
    const isToday = dateKey(dd) === dateKey(now);
    const isDone = !!checkedDays[key];
    const isRest = dow === 0;
    const isFuture = dd > now;

    let cls = 'cal-cell';
    if (isToday) cls += ' today-cell';
    if (isDone) cls += ' done-cell';
    if (isRest && !isDone) cls += ' rest-cell';

    let inner = d;
    if (isDone) inner = `<span style="font-size:16px">✓</span>`;
    else if (isRest) inner = `${d}<br><span style="font-size:8px">REST</span>`;

    html += `<div class="${cls}" onclick="${!isFuture ? `toggleCalCheck('${key}')`:''}" style="${isFuture?'opacity:0.3;cursor:default;':''}">${inner}</div>`;
  }

  // Fill rest of grid
  const totalCells = firstDay + daysInMonth;
  const remaining = totalCells % 7 === 0 ? 0 : 7 - (totalCells % 7);
  for (let i = 1; i <= remaining; i++) {
    html += `<div class="cal-cell other-month">${i}</div>`;
  }

  html += `</div>`;

  // Legend
  html += `
    <div style="display:flex; gap:12px; flex-wrap:wrap; font-size:12px; color:var(--muted); margin-bottom:16px;">
      <div style="display:flex;align-items:center;gap:6px;"><div style="width:14px;height:14px;border-radius:4px;background:var(--rose);"></div>完了</div>
      <div style="display:flex;align-items:center;gap:6px;"><div style="width:14px;height:14px;border-radius:4px;background:var(--light-sage);"></div>休養日</div>
      <div style="display:flex;align-items:center;gap:6px;"><div style="width:14px;height:14px;border-radius:4px;border:1.5px solid var(--rose);"></div>今日</div>
    </div>
    <div style="font-size:12px; color:var(--muted); text-align:center; padding:8px;">過去の日付をタップして完了/未完了を切り替えられます</div>
  `;

  document.getElementById('page-tracker').innerHTML = html;
}

function changeMonth(dir) {
  trackerMonth = new Date(trackerMonth.getFullYear(), trackerMonth.getMonth() + dir, 1);
  renderTracker();
}

function toggleCalCheck(key) {

  localStorage.setItem('wbt_checked', JSON.stringify(checkedDays));
  renderTracker();
}

function addEvent() {
  const name = document.getElementById('ev-name').value.trim();
  const date = document.getElementById('ev-date').value;
  if (!name || !date) { alert('名前と日付を入力してください'); return; }
  customEvents.push({ name, date, type: 'custom', fixed: false });
  localStorage.setItem('wbt_events', JSON.stringify(customEvents));
  renderEvents();
}

function deleteEvent(idx) {
  if (!confirm('この予定を削除しますか？')) return;
  customEvents.splice(idx, 1);
  localStorage.setItem('wbt_events', JSON.stringify(customEvents));
  renderEvents();
}
